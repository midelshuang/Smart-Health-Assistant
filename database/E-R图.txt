    user_profiles {
        varchar user_id PK
        jsonb health_tags

        bigint id PK
        varchar user_id FK
        int nutrition_id FK

    nutrition_facts {
        int id PK

        decimal calories_per_100g

        varchar data_source
    }

        int id PK
        varchar food_name
        decimal gi_value
        varchar food_group
    }

    chat_histories {
        int id PK
        uuid message_uuid UK
        varchar session_id
        text message_content
        jsonb metadata
    }

    user_feedbacks {
        int id PK
        uuid message_id FK
        smallint rating
        varchar feedback_type
    }

    fitness_goals {
        bigint id PK
        varchar user_id FK
        varchar goal_type

        varchar status
    }
